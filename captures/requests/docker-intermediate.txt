# Capture requests for Part C (Docker, intermediate). One line per capture:
#   <part>/<id> | <command run from repo root>
# All are self-contained and clean up after themselves. API_PORT is moved off
# 8080 because the kind lab maps host 8080.

docker-intermediate/compose-config | docker compose -f examples/compose/compose.yaml config | head -40
docker-intermediate/compose-services-default | docker compose -f examples/compose/compose.yaml config --services | sort
docker-intermediate/compose-services-profiles | docker compose -f examples/compose/compose.yaml --profile load --profile observability config --services | sort
docker-intermediate/compose-up-wait | cd examples/compose && API_PORT=18080 docker compose up -d --wait 2>&1 | tail -8; API_PORT=18080 docker compose down >/dev/null 2>&1
docker-intermediate/compose-ps-health | cd examples/compose && API_PORT=18080 docker compose up -d --wait >/dev/null 2>&1; API_PORT=18080 docker compose ps --format 'table {{.Service}}\t{{.Status}}'; API_PORT=18080 docker compose down >/dev/null 2>&1
docker-intermediate/compose-scale-worker | cd examples/compose && API_PORT=18080 docker compose up -d --wait >/dev/null 2>&1; API_PORT=18080 docker compose up -d --scale worker=3 --no-recreate 2>&1 | tail -4; API_PORT=18080 docker compose ps --format 'table {{.Service}}\t{{.Name}}' | grep worker; API_PORT=18080 docker compose down >/dev/null 2>&1
docker-intermediate/history-naive | docker build -q -t di-naive -f examples/dockerfiles/size-naive-vs-optimised/Dockerfile.naive examples/dockerfiles/size-naive-vs-optimised >/dev/null && docker history --no-trunc --format 'table {{.CreatedBy}}\t{{.Size}}' di-naive | head -20
docker-intermediate/history-optimised | docker build -q -t di-optimised -f examples/dockerfiles/size-naive-vs-optimised/Dockerfile.optimised examples/dockerfiles/size-naive-vs-optimised >/dev/null && docker history --format 'table {{.CreatedBy}}\t{{.Size}}' di-optimised | head -20
docker-intermediate/size-images | docker build -q -t di-naive -f examples/dockerfiles/size-naive-vs-optimised/Dockerfile.naive examples/dockerfiles/size-naive-vs-optimised >/dev/null; docker build -q -t di-optimised -f examples/dockerfiles/size-naive-vs-optimised/Dockerfile.optimised examples/dockerfiles/size-naive-vs-optimised >/dev/null; docker images --format 'table {{.Repository}}\t{{.Tag}}\t{{.Size}}' | grep -E 'di-naive|di-optimised|REPOSITORY'
docker-intermediate/cache-unordered-rebuild | docker build -q -t di-unordered -f examples/dockerfiles/cache-ordering/Dockerfile.unordered examples/dockerfiles/cache-ordering >/dev/null; echo '# print("touched")' >> examples/dockerfiles/cache-ordering/app.py; docker build -t di-unordered -f examples/dockerfiles/cache-ordering/Dockerfile.unordered examples/dockerfiles/cache-ordering 2>&1 | grep -iE 'RUN|pip|CACHED|=> ' | head -12; git checkout examples/dockerfiles/cache-ordering/app.py 2>/dev/null
docker-intermediate/cache-ordered-rebuild | docker build -q -t di-ordered -f examples/dockerfiles/cache-ordering/Dockerfile.ordered examples/dockerfiles/cache-ordering >/dev/null; echo '# print("touched")' >> examples/dockerfiles/cache-ordering/app.py; docker build -t di-ordered -f examples/dockerfiles/cache-ordering/Dockerfile.ordered examples/dockerfiles/cache-ordering 2>&1 | grep -iE 'pip|CACHED|=> ' | head -12; git checkout examples/dockerfiles/cache-ordering/app.py 2>/dev/null
docker-intermediate/init-ps-noinit | docker build -q -t di-noinit -f examples/dockerfiles/init-and-zombies/Dockerfile.noinit examples/dockerfiles/init-and-zombies >/dev/null && docker run -d --name di-noinit di-noinit >/dev/null 2>&1; sleep 2; docker exec di-noinit ps -o pid,ppid,args 2>/dev/null | head; docker rm -f di-noinit >/dev/null 2>&1
docker-intermediate/init-ps-tini | docker build -q -t di-tini -f examples/dockerfiles/init-and-zombies/Dockerfile.tini examples/dockerfiles/init-and-zombies >/dev/null && docker run -d --name di-tini di-tini >/dev/null 2>&1; sleep 2; docker exec di-tini ps -o pid,ppid,args 2>/dev/null | head; docker rm -f di-tini >/dev/null 2>&1
docker-intermediate/signal-stop-shell | docker build -q -t di-shell -f examples/dockerfiles/signal-forms/Dockerfile.shell examples/dockerfiles/signal-forms >/dev/null && docker run -d --name di-shell di-shell >/dev/null 2>&1; ( time docker stop di-shell >/dev/null ) 2>&1 | grep real; docker rm -f di-shell >/dev/null 2>&1
docker-intermediate/signal-stop-exec | docker build -q -t di-exec -f examples/dockerfiles/signal-forms/Dockerfile.exec examples/dockerfiles/signal-forms >/dev/null && docker run -d --name di-exec di-exec >/dev/null 2>&1; ( time docker stop di-exec >/dev/null ) 2>&1 | grep real; docker rm -f di-exec >/dev/null 2>&1
docker-intermediate/network-inspect-backend | docker network create di-backend --internal >/dev/null 2>&1; docker network inspect di-backend --format '{{.Name}} internal={{.Internal}} driver={{.Driver}} subnet={{range .IPAM.Config}}{{.Subnet}}{{end}}'; docker network rm di-backend >/dev/null 2>&1
docker-intermediate/dns-resolution | docker network create di-dns >/dev/null 2>&1; docker run -d --name di-db --network di-dns busybox:1.37-musl sleep 300 >/dev/null 2>&1; docker run --rm --network di-dns busybox:1.37-musl nslookup di-db 2>&1 | head -8; docker rm -f di-db >/dev/null 2>&1; docker network rm di-dns >/dev/null 2>&1
