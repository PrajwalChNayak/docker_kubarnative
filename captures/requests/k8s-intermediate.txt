k8s-intermediate/qos-classes | kubectl -n tasklane get pods -o custom-columns='NAME:.metadata.name,QOS:.status.qosClass,NODE:.spec.nodeName,CPU_REQ:.spec.containers[0].resources.requests.cpu,MEM_LIM:.spec.containers[0].resources.limits.memory'

k8s-intermediate/gateway-describe | kubectl -n tasklane describe gateway tasklane

k8s-intermediate/curl-https | kubectl -n tasklane get secret tasklane-tls -o jsonpath='{.data.ca\.crt}' | base64 -d > /tmp/tasklane-ca.crt && curl -sS -i --cacert /tmp/tasklane-ca.crt --resolve tasklane.localhost:8443:127.0.0.1 https://tasklane.localhost:8443/
k8s-intermediate/curl-redirect | curl -sS -o /dev/null -D - --resolve tasklane.localhost:8080:127.0.0.1 http://tasklane.localhost:8080/
k8s-intermediate/curl-split | for i in $(seq 1 20); do curl -s --cacert /tmp/tasklane-ca.crt --resolve tasklane.localhost:8443:127.0.0.1 https://tasklane.localhost:8443/; echo; done | sort | uniq -c
k8s-intermediate/curl-canary-header | curl -sS -i -H 'x-tasklane-track: canary' --cacert /tmp/tasklane-ca.crt --resolve tasklane.localhost:8443:127.0.0.1 https://tasklane.localhost:8443/
k8s-intermediate/openssl-cert | openssl s_client -connect 127.0.0.1:8443 -servername tasklane.localhost -CAfile /tmp/tasklane-ca.crt </dev/null 2>/dev/null | openssl x509 -noout -subject -issuer -dates -ext subjectAltName
